  try {

  c.getConstructor(Context.class).newInstance(...));

      // 静默吞掉
  } catch (InstantiationException ie) {
      // 静默吞掉
  } catch (IllegalAccessException ie) {
      // 静默吞掉
  } catch (InvocationTargetException ite) {
      // 静默吞掉 — getReadableDatabase() 的异常走这里

  目录，然后重新编译即可。同样需要检查其他本地词典（如
  Ode2）的数据库文件是否也缺失。
